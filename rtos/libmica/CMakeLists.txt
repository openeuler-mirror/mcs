cmake_minimum_required(VERSION 3.16)
project(mica LANGUAGES C)

include(GNUInstallDirs)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/config.cmake")
  include("${CMAKE_CURRENT_SOURCE_DIR}/config.cmake")
endif()
if(NOT MICA_PED MATCHES "^(baremetal|hetero|xen)$")
  message(FATAL_ERROR "MICA_PED must be baremetal, hetero or xen")
endif()
if(SUPPORT_RPC OR WITH_PROXY OR WITH_SHARED_LIB)
  message(FATAL_ERROR "RPC/proxy/shared builds are not supported by this runtime contract")
endif()
if(MICA_PED STREQUAL "xen")
  if(NOT MICA_XEN_PROVIDER MATCHES "^(builtin|native)$")
    message(FATAL_ERROR "MICA_XEN_PROVIDER must be builtin or native for MICA_PED=xen")
  endif()
  if(MICA_XEN_PROVIDER STREQUAL "builtin")
    set(CMAKE_ASM_COMPILER "${CMAKE_C_COMPILER}")
    enable_language(ASM)
  endif()
endif()

add_subdirectory(lib)
add_subdirectory(src)
